```java
private AprilTagFieldLayout loadFieldLayout() {
    try {
        // Option 1: Load from WPILib built-in (recommended for 2026)
        return AprilTagFieldLayout.loadField(AprilTagFields.k2026Reefscape);
        
    } catch (Exception e) {
        System.err.println("ERROR: Could not load 2026 field layout");
        e.printStackTrace();
        
        // Option 2: Try loading from deploy directory
        try {
            return new AprilTagFieldLayout(
                Filesystem.getDeployDirectory().toPath()
                    .resolve("2026-apriltags.json")
            );
        } catch (IOException ex) {
            throw new RuntimeException("Failed to load AprilTag layout", ex);
        }
    }
}
```
